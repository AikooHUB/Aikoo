-- Fungsi bantu untuk mengecek apakah part berwarna emas
local function isGoldColor(part)
    local color = part.Color
    if color.R > 0.7 and color.G > 0.5 and color.B < 0.3 then
        return true
    end
    
    local bColor = part.BrickColor.Name
    if bColor == "Bright yellow" or bColor == "New Yeller" or bColor == "Deep orange" or bColor == "Neon orange" then
        return true
    end
    
    return false
end

-- 1. Logika Auto Gold (Loop)
CollectToggleBtn.MouseButton1Click:Connect(function()
    isAutoCollectActive = not isAutoCollectActive
    
    if isAutoCollectActive then
        CollectToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 127)
        CollectToggleBtn.Text = "Auto Gold: ON"
        StatusLabel.Text = "Status: Memulai Auto Gold (Loop)..."
        
        task.spawn(function()
            while isAutoCollectActive do
                local character = LocalPlayer.Character
                if not character or not character:FindFirstChild("HumanoidRootPart") then
                    StatusLabel.Text = "Status: Karakter tidak ada!"
                    task.wait(1)
                    continue
                end
                
                local hrp = character.HumanoidRootPart
                
                StatusLabel.Text = "Status: TP ke titik awal..."
                hrp.CFrame = CFrame.new(495.53, 15.20, 958.38)
                task.wait(1.0)
                
                if not isAutoCollectActive then break end
                
                StatusLabel.Text = "Status: Bersiap mengambil item emas..."
                task.wait(0.5)
                
                for _, folder in ipairs(Workspace:GetChildren()) do
                    if not isAutoCollectActive then break end
                    if folder:IsA("Folder") and folder.Name:match("^LocalCollectibles_") then
                        for _, item in ipairs(folder:GetChildren()) do
                            if not isAutoCollectActive then break end
                            
                            if item:IsA("BasePart") and isGoldColor(item) then
                                StatusLabel.Text = "Status: Mengambil item emas (" .. item.Name .. ")"
                                hrp.CFrame = item.CFrame + Vector3.new(0, 3, 0)
                                task.wait(0.2)
                                
                                local prompt = item:FindFirstChildOfClass("ProximityPrompt")
                                if prompt then
                                    pcall(function() fireproximityprompt(prompt) end)
                                else
                                    local touchInterest = item:FindFirstChild("TouchInterest")
                                    if touchInterest then
                                        firetouchinterest(hrp, item, 0)
                                        task.wait(0.1)
                                        firetouchinterest(hrp, item, 1)
                                    end
                                end
                                task.wait(0.15)
                            end
                        end
                    end
                end
                
                if not isAutoCollectActive then break end
                
                StatusLabel.Text = "Status: Selesai putaran ini, pindah ke Koor Akhir..."
                task.wait(0.5)
                
                hrp.CFrame = CFrame.new(491.18, 46.30, -5056.67)
                StatusLabel.Text = "Status: Menunggu putaran berikutnya..."
                task.wait(1.5)
            end
            
            CollectToggleBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
            CollectToggleBtn.Text = "Auto Gold: OFF"
            StatusLabel.Text = "Status: Auto Gold dimatikan"
        end)
        
    else
        CollectToggleBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
        CollectToggleBtn.Text = "Auto Gold: OFF"
        StatusLabel.Text = "Status: Auto Gold dimatikan"
    end
end)

-- 2. Logika Auto Island (Teleport Saja)
IslandToggleBtn.MouseButton1Click:Connect(function()
    isAutoIslandActive = not isAutoIslandActive
    
    if isAutoIslandActive then
        IslandToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 127)
        IslandToggleBtn.Text = "Auto Island TP: ON"
        StatusLabel.Text = "Status: Memulai Auto Island TP..."
        
        task.spawn(function()
            while isAutoIslandActive do
                local character = LocalPlayer.Character
                if not character or not character:FindFirstChild("HumanoidRootPart") then
                    StatusLabel.Text = "Status: Karakter tidak ada!"
                    task.wait(1)
                    continue
                end
                
                local hrp = character.HumanoidRootPart
                
                local playerID = nil
                for _, folder in ipairs(Workspace:GetChildren()) do
                    if folder:IsA("Folder") and folder.Name:match("^LocalCollectibles_") then
                        playerID = folder.Name:match("_(%d+)$")
                        break
                    end
                end
                
                local islandsFolder = Workspace:FindFirstChild("Islands")
                if islandsFolder and playerID then
                    local targetIslandName = "Island_" .. playerID
                    local island = islandsFolder:FindFirstChild(targetIslandName)
                    
                    if island then
                        for _, block in ipairs(island:GetChildren()) do
                            if not isAutoIslandActive then break end
                            
                            if block:IsA("BasePart") then
                                StatusLabel.Text = "Status: TP ke " .. block.Name
                                hrp.CFrame = block.CFrame + Vector3.new(0, 3, 0)
                                task.wait(0.2)
                            end
                        end
                        StatusLabel.Text = "Status: Selesai satu putaran Island TP"
                    else
                        StatusLabel.Text = "Status: Island_" .. playerID .. " tidak ditemukan!"
                    end
                else
                    StatusLabel.Text = "Status: Gagal mendeteksi Player ID/Folder Islands!"
                end
                
                task.wait(1)
            end
            
            IslandToggleBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
            IslandToggleBtn.Text = "Auto Island TP: OFF"
            StatusLabel.Text = "Status: Auto Island TP dimatikan"
        end)
        
    else
        IslandToggleBtn.BackgroundColor3 = Color3.fromRGB(180, 50, 50)
        IslandToggleBtn.Text = "Auto Island TP: OFF"
        StatusLabel.Text = "Status: Auto Island TP dimatikan"
    end
end)
