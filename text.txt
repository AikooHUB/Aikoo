-- 2. FUNGSI DRAG, MINIMIZE & CLOSE GUI
-- ==========================================
local dragging, dragInput, dragStart, startPos
topBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
topBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Murni mencari tombol Replay yang sedang terbuka
local function getVisibleReplayButton()
    local playerGui = player:FindFirstChild("PlayerGui")
    if not playerGui then return nil end
    
    for _, guiObj in pairs(playerGui:GetDescendants()) do
        if guiObj:IsA("GuiButton") and guiObj.Name == "ReplayButton" then
            if guiObj.Parent and guiObj.Parent.Visible then
                return guiObj
            end
        end
    end
    return nil
end

-- Melakukan aksi klik paksa pada tombol yang ditemukan
local function clickButton(targetBtn)
    pcall(function()
        if type(firesignal) == "function" then
            firesignal(targetBtn.MouseButton1Up)
            firesignal(targetBtn.MouseButton1Click)
            firesignal(targetBtn.Activated)
        elseif type(getconnections) == "function" then
            for _, conn in pairs(getconnections(targetBtn.MouseButton1Up)) do
                if type(conn.Function) == "function" then conn:Function() end
            end
        end
        
        local inset, _ = GuiService:GetGuiInset()
        local absPos = targetBtn.AbsolutePosition
        local absSize = targetBtn.AbsoluteSize
        local x = absPos.X + (absSize.X / 2)
        local y = absPos.Y + (absSize.Y / 2) + inset.Y
        
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 1)
        task.wait(0.05)
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 1)
    end)
end

-- Fungsi Utama Loop
local function autoObbyLoop()
    while isRunning do
        -- 1. SETUP KARAKTER
        local character = player.Character or player.CharacterAdded:Wait()
        local hrp = character:WaitForChild("HumanoidRootPart", 5)
        
        if not hrp then
            task.wait(1)
            continue
        end
        
        local totalCP = #tpPoints
        local TARGET_DURATION = math.random(131, 141)
        local startTime = tick()
        
        -- 2. TWEEN KE AWAL
        statusLabel.Text = "Status: Tweening ke titik awal..."
        local tweenInfo = TweenInfo.new(3, Enum.EasingStyle.Linear)
        local tween = TweenService:Create(hrp, tweenInfo, {CFrame = CFrame.new(490, 282, 610)})
        tween:Play()
        tween.Completed:Wait()
        
        -- 3. TP LOOP KE SEMUA CP
        for i, point in ipairs(tpPoints) do
            if not isRunning then break end
            statusLabel.Text = "Progress CP: " .. i .. " / " .. totalCP
            
            local randomDelay = math.random(15, 20) / 10
            task.wait(randomDelay)
            
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                player.Character.HumanoidRootPart.CFrame = CFrame.new(point)
            else
                break
            end
        end
        
        -- 4. WAITING TIME
        local elapsedTime = tick() - startTime
        local timeLeft = TARGET_DURATION - elapsedTime
        
        while timeLeft > 0 and isRunning do
            statusLabel.Text = "Sisa Waktu: " .. math.floor(timeLeft) .. " Detik"
            task.wait(1)
            elapsedTime = tick() - startTime
            timeLeft = TARGET_DURATION - elapsedTime
        end
        
        if not isRunning then break end
        
        -- 5. TP KE PORTAL
        if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            statusLabel.Text = "Status: Teleporting ke Portal!"
            local teleportTarget = game:GetService("Workspace"):FindFirstChild("Utility")
            
            if teleportTarget and teleportTarget:FindFirstChild("Teleporters") and teleportTarget.Teleporters:FindFirstChild("Portal_to_Wc") and teleportTarget.Teleporters.Portal_to_Wc:FindFirstChild("Collision") then
                local portalCollision = teleportTarget.Teleporters.Portal_to_Wc.Collision
                player.Character.HumanoidRootPart.CFrame = portalCollision.CFrame
                statusLabel.Text = "Status: Mencapai Portal!"
            else
                statusLabel.Text = "Error: Portal tidak ditemukan!"
            end
        end
        
        -- 6. MENUNGGU TOMBOL MUNCUL & KLIK SAMPAI HILANG (KONFIRMASI)
        statusLabel.Text = "Status: Menunggu Tombol Replay..."
        local activeReplayBtn = nil
        
        -- Loop A: Tunggu sampai UI Replay benar-benar muncul di layar
        while isRunning do
            activeReplayBtn = getVisibleReplayButton()
            if activeReplayBtn then break end
            task.wait(0.5)
        end
        
        if not isRunning then break end
        
        -- Loop B: Tekan terus tombolnya SESEKALI sampai panelnya disembunyikan gamenya
        statusLabel.Text = "Status: Mencoba tekan Replay..."
        while isRunning and activeReplayBtn and activeReplayBtn.Parent and activeReplayBtn.Parent.Visible do
            clickButton(activeReplayBtn)
            task.wait(1) -- Beri jeda 1 detik tiap kali nembak klik, supaya ngga spamming server
        end
        
        if not isRunning then break end
        
        statusLabel.Text = "Status: Replay Berhasil!"
        
        -- 7. WARM UP SEBELUM LOOP BERIKUTNYA
        -- Timer ini HANYA akan jalan kalau Loop B di atas sudah tembus (tombol hilang karena berhasil diklik)
        local warmupTime = math.random(20, 50) / 10
        local timer = warmupTime
        
        while timer > 0 and isRunning do
            statusLabel.Text = string.format("Status: Warm up %.1f detik...", timer)
            task.wait(0.1)
            timer = timer - 0.1
        end
    end
    
    -- JIKA LOOP DIHENTIKAN OLEH USER
    startBtn.Text = "Start Auto Obby"
    startBtn.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    startBtn.TextColor3 = Color3.fromRGB(220, 220, 220)
    statusLabel.Text = "Status: Idle"
end

-- ==========================================
-- 4. TOMBOL TOGGLE (START / STOP)
-- ==========================================
startBtn.MouseButton1Click:Connect(function()
    if isRunning then
        isRunning = false
        startBtn.Text = "Stopping Loop..."
        startBtn.BackgroundColor3 = Color3.fromRGB(150, 100, 30)
        statusLabel.Text = "Status: Membatalkan urutan..."
    else
        isRunning = true
        startBtn.Text = "Stop Auto Obby"
        startBtn.BackgroundColor3 = Color3.fromRGB(150, 40, 40)
        startBtn.TextColor3 = Color3.fromRGB(255, 200, 200)
        task.spawn(autoObbyLoop)
    end
end)
